# The frozen Pico Full build profile (pico2 P1-F2, ADR-0012 point 3, research D33 Q3/Q4; object sets P1-R4, research D58):
# board pico2, platform rp2350-riscv, Debug = -Og plus -g. Four object sets, each a pair of programs built with the
# WebAssembly clang:
#   base_<set> -- pico_stdlib (and the set's libraries): its objects are linked whole into every student program built
#                 with that set (as CMake links them; objects registering through sections must not sit in an archive)
#   full_<set> -- the same plus every further library of the pack: objects of sources that base_<set> does not compile
#                 are the set's lazily linked objects (an archive's semantics), so ld.lld takes only what a program uses.
# Sets (P1-R4 design, docs/architecture/pico2-p1r4-pack.md):
#   uart    -- stdio over UART only (the default; what P1-F2 shipped)
#   uart_mc -- uart plus pico_multicore (a project whose files include "pico/multicore.h")
#   usb     -- stdio over UART and USB (pico_stdio_usb, TinyUSB's device stack, pico_usb_reset; the project option
#              "UART and USB", P1-F3 Q3 A)
#   usb_mc  -- usb plus pico_multicore
# build-pack.mjs checks, per set, that the sources both programs compile give identical code (the further libraries'
# LIB_* defines change nothing shared); LIB_PICO_MULTICORE and LIB_PICO_STDIO_USB do change shared code, which is why
# they make sets of their own instead of joining the further libraries.
# Left out (build-pack.mjs lists them with the reason in the pack manifest):
#   pico_low_power                       -- its LIB_* define changes shared code (runtime init, sections)
#   pico_stdio_rtt, pico_stdio_semihosting -- no simulated debug probe
#   pico_bootsel_via_double_reset        -- runs from a constructor; an archive member that nothing references is never linked
#   pico_cyw43_*, pico_btstack, pico_lwip, pico_mbedtls -- radio and network (no radio on pico2)
#   hardware_dcp, hardware_rcp, cmsis    -- Arm only
# pico_status_led (P1-R4): a further library; its PIO LED driver's header is generated by pioasm, built on the host with
# the MSYS2 g++ (build-profile.ps1 puts it on PATH).
# Copyright (c) 2026 Sajid Muhaimin Choudhury. Licensed under the Apache License, Version 2.0.
cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2 CACHE STRING "" FORCE)
set(PICO_PLATFORM rp2350-riscv CACHE STRING "" FORCE)
include($ENV{PICO_SDK_PATH}/pico_sdk_init.cmake)
project(pico_full_profile C CXX ASM)
set(CMAKE_C_STANDARD 11)

# -Og -g (the SDK's Debug is -Og without -g); host paths of the SDK and this build are replaced in debug information and
# __FILE__ by the guest paths the pack uses
# (2026-11-14) also the sysroot and clang's resource directory, whose include directories were left in the first pack's
# debug information as host paths. Set before pico_sdk_init(), which adds the SDK's own libraries
# (boot_stage2's bs2_default_library kept the build folder's path when these came after it).
file(TO_CMAKE_PATH "$ENV{PICO_FULL_WORK}" _work)
foreach(LANG IN ITEMS C CXX ASM)
  set(CMAKE_${LANG}_FLAGS_DEBUG "-Og -g")
  string(APPEND CMAKE_${LANG}_FLAGS " -ffile-prefix-map=$ENV{PICO_SDK_PATH}=/pico/sdk -ffile-prefix-map=${CMAKE_BINARY_DIR}=/pico/build")
  string(APPEND CMAKE_${LANG}_FLAGS " -ffile-prefix-map=${_work}/build/sysroot-rv32=/pico/sysroot -ffile-prefix-map=${_work}/build/clang-wasm/lib/clang/23=/pico/clang")
endforeach()
# The build date in binary info (P1-R4, research D58: a fixed date string instead of none): the pack's date, from
# SOURCE_DATE_EPOCH (which string(TIMESTAMP) honours), in __DATE__'s form "Mmm dd yyyy" (day padded with a space), so
# that the pack is the same on every day it is rebuilt. Every program built from the pack carries this date.
include(${CMAKE_CURRENT_LIST_DIR}/../cmake/build-date.cmake)
pico_full_build_date(_build_date)
add_compile_definitions(PICO_PROGRAM_BUILD_DATE="${_build_date}")
pico_sdk_init()

set(PACK_EXTRA
  hardware_adc hardware_boot_lock hardware_dma hardware_exception hardware_flash hardware_hazard3 hardware_i2c
  hardware_interp hardware_pio hardware_powman hardware_pwm hardware_riscv_platform_timer hardware_rosc hardware_sha256
  hardware_spi hardware_vreg hardware_watchdog hardware_xip_cache
  pico_aon_timer pico_async_context_poll pico_async_context_threadsafe_background pico_atomic pico_bootrom pico_flash
  pico_i2c_slave pico_rand pico_sha256 pico_status_led pico_unique_id pico_util)
set(PACK_LINKED "")
set(PACK_MISSING "")
foreach(LIB IN LISTS PACK_EXTRA)
  if(TARGET ${LIB})
    list(APPEND PACK_LINKED ${LIB})
  else()
    list(APPEND PACK_MISSING ${LIB})
  endif()
endforeach()

set(PACK_SETS uart uart_mc usb usb_mc)
foreach(SET IN LISTS PACK_SETS)
  set(SET_LIBS pico_stdlib)
  if(SET MATCHES "_mc$")
    list(APPEND SET_LIBS pico_multicore)
  endif()
  set(USB 0)
  if(SET MATCHES "^usb")
    set(USB 1)
  endif()
  add_executable(base_${SET} main.c)
  target_link_libraries(base_${SET} ${SET_LIBS})
  pico_enable_stdio_uart(base_${SET} 1)
  pico_enable_stdio_usb(base_${SET} ${USB})
  # a static library: CMake compiles the libraries' sources into it and never links it (the pack builder fixes objects
  # for lld first, see build-pack.mjs)
  add_library(full_${SET} STATIC main.c)
  target_link_libraries(full_${SET} ${SET_LIBS} ${PACK_LINKED})
  pico_enable_stdio_uart(full_${SET} 1)
  pico_enable_stdio_usb(full_${SET} ${USB})
endforeach()
string(REPLACE ";" " " _sets "${PACK_SETS}")
file(WRITE ${CMAKE_BINARY_DIR}/pack-libraries.json "{\"linked\": \"${PACK_LINKED}\", \"missing\": \"${PACK_MISSING}\", \"sets\": \"${_sets}\", \"buildDate\": \"${_build_date}\"}\n")
